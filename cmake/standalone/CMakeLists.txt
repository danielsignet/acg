cmake_minimum_required(VERSION 3.21)
project(vrlib VERSION 2.0 LANGUAGES C CXX)
get_filename_component(VRLIB_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build configuration" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${VRLIB_SOURCE_DIR}/dependencies/glfw-3.3.9" "dependencies/glfw-3.3.9" EXCLUDE_FROM_ALL)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_COLLADA_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
add_subdirectory("${VRLIB_SOURCE_DIR}/dependencies/assimp-5.3.1" "dependencies/assimp-5.3.1" EXCLUDE_FROM_ALL)
add_subdirectory("${VRLIB_SOURCE_DIR}/dependencies/freetype-2.3.11" "dependencies/freetype-2.3.11" EXCLUDE_FROM_ALL)
find_package(OpenGL REQUIRED)
include(CTest)
add_subdirectory("${VRLIB_SOURCE_DIR}/applications" "applications")
if(MSVC)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
endif()
if(BUILD_TESTING)
  add_executable(culling_tests "${VRLIB_SOURCE_DIR}/tests/culling_tests.cpp")
  target_link_libraries(culling_tests PRIVATE vrlib)
  add_test(NAME culling COMMAND culling_tests)
  add_executable(portal_tests "${VRLIB_SOURCE_DIR}/tests/portal_tests.cpp")
  target_link_libraries(portal_tests PRIVATE vrlib)
  add_test(NAME portals COMMAND portal_tests)
  if(EXISTS "${VRLIB_SOURCE_DIR}/tests/scenegraph_tests.cpp")
    add_executable(scenegraph_tests "${VRLIB_SOURCE_DIR}/tests/scenegraph_tests.cpp")
    target_link_libraries(scenegraph_tests PRIVATE vrlib)
    add_test(NAME scenegraph COMMAND scenegraph_tests "${VRLIB_SOURCE_DIR}")
  endif()
  add_test(NAME rendering COMMAND viewer --hidden --validate --frames 4
    --dot "${PROJECT_BINARY_DIR}/artifacts/scenegraph.dot")
endif()
